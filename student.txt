#!/bin/bash
echo "Enter marks of 5 subjects:"
read m1 m2 m3 m4 m5
total=$((m1+m2+m3+m4+m5))
avg=$((total/5))
echo "Total = $total"
echo "Average = $avg"
if [ $avg -ge 90 ]; then
echo "Grade: A"
elif [ $avg -ge 75 ]; then
echo "Grade: B"
elif [ $avg -ge 50 ]; then
echo "Grade: C"
else
echo "Grade: Fail"
fi
